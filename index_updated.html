window.calculateCGPA = () => {
let totalCredits = 0, totalPoints = 0, hasValid = false;
document.querySelectorAll('.cgpa-row').forEach(row => {
const cred = parseFloat(row.querySelector('.cgpa-credit').value);
const grade = parseFloat(row.querySelector('.cgpa-grade').value);
if (cred > 0 && !isNaN(grade)) {
hasValid = true; totalCredits += cred; totalPoints += (cred * grade);
}
});
document.getElementById('cgpaDisplay').textContent = hasValid && totalCredits > 0 ? (totalPoints / totalCredits).toFixed(2) : '0.00';
if (!hasValid) window.showToast('Please enter valid credits and grades.', 'error');
}
window.showToast = (message, type = 'success') => {
const container = document.getElementById('toastContainer');
const toast = document.createElement('div');
let conf = type === 'success' ? 'border-green-600 fa-circle-check text-green-600' : (type === 'error' ? 'border-red-600 fa-circle-exclamation text-red-600' : 'border-stone-600 fa-circle-info text-stone-600');
toast.className = `toast-enter flex items-center gap-3 px-5 py-4 rounded-xl shadow-lg border border-stone-100 bg-white border-l-4 ${conf.split(' ')[0]} pointer-events-auto`;
toast.innerHTML = `<i class="fa-solid ${conf.split(' ').slice(1).join(' ')} text-xl"></i> <span class="font-semibold text-stone-700 text-sm">${message}</span>`;
container.appendChild(toast);
setTimeout(() => { toast.style.opacity = '0'; setTimeout(() => toast.remove(), 300); }, 3000);
}

async function init() {
try {
// Must authenticate before querying database
await signInAnonymously(auth);
onSnapshot(query(collection(db, COLLECTION_PATH)), (snapshot) => {
let newFs = { 'root': { name: 'Home', type: 'folder', children: [] } };
snapshot.forEach(doc => newFs[doc.id] = { id: doc.id, ...doc.data(), children: [] });
snapshot.forEach(doc => {
const pid = doc.data().parentId || 'root';
if (newFs[pid]) newFs[pid].children.push(doc.id);
});
fileSystem = newFs;
renderExplorer();
});
} catch (err) {
window.showToast('Firebase connection failed. Check console.', 'error');
console.error(err);
}
}
